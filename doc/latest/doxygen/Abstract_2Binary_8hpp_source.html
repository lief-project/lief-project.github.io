240 template<class T>
241 LIEF::result<T>
242 get_int_from_virtual_address(uint64_t va,
243 VA_TYPES addr_type = VA_TYPES::AUTO) const {
244 T value;
245 static_assert(std::is_integral_v<T>, "Require an integral type");
246 span<const uint8_t> raw =
247 get_content_from_virtual_address(va, sizeof(T), addr_type);
248 if (raw.empty() || raw.size() < sizeof(T)) {
249 return make_error_code(lief_errors::read_error);
250 }
251
252 std::copy(raw.data(), raw.data() + sizeof(T),
253 reinterpret_cast<uint8_t*>(&value));
254 return value;
255 }
