120 class ITERATOR_T = typename decay_t<T>::iterator>
121class ref_iterator {
122 template<class, typename, class>
123 friend class ref_iterator;
124
125 public:
126 using iterator_category = std::bidirectional_iterator_tag;
127 using iterator_concept =
128 details::iterator_concept_t<T, U, std::bidirectional_iterator_tag>;
129 using value_type = std::remove_cv_t<std::remove_pointer_t<decay_t<U>>>;
130 using difference_type = ptrdiff_t;
131 using pointer = std::remove_pointer_t<U>*;
132 using reference = std::remove_pointer_t<U>&;
133
134 using container_type = T; // e.g. std::vector<Section*>&
135 using DT_VAL = U; // e.g. Section*
136 using DT = decay_t<T>; // e.g. std::vector<Section>
137 using ref_t = typename ref_iterator::reference;
138 using pointer_t = typename ref_iterator::pointer;
139
140 ref_iterator() = default;
141
142 ref_iterator(const ref_iterator& other) :
143 container_{other.container_},
144 it_{other.it_},
145 distance_{other.distance_} {
146 rebase();
147 }
148
149 ref_iterator(ref_iterator&& other) noexcept :
150 container_{std::move(other.container_)},
151 it_{other.it_},
152 distance_{other.distance_} {
153 rebase();
154 other.container_ = {};
155 other.it_ = {};
156 other.distance_ = 0;
157 }
158
159 ref_iterator& operator=(const ref_iterator& other) LIEF_LIFETIMEBOUND {
160 ref_iterator copy{other};
161 swap(copy);
162 return *this;
163 }
164
165 ref_iterator& operator=(ref_iterator&& other) noexcept LIEF_LIFETIMEBOUND {
166 swap(other);
167 return *this;
168 }

170 template<class T2, typename U2, class IT2,
171 typename = std::enable_if_t<
172 !std::is_same_v<ref_iterator, ref_iterator<T2, U2, IT2>> &&
173 std::is_same_v<decay_t<T>, decay_t<T2>>
174 >>
175 ref_iterator(const ref_iterator<T2, U2, IT2>& other) :
176 distance_{other.distance_} {
177 if (auto* container = other.container_.get(); container != nullptr) {
178 if constexpr (std::is_reference_v<T>) {
179 container_ = details::container_storage<T>{const_cast<T>(*container)};
180 } else {
181 container_ = details::container_storage<T>{*container};
182 }
183 it_ = std::next(std::begin(*container_.get()), distance_);
184 }
185 }
186
187 ~ref_iterator() = default;
188
189 // NOLINTNEXTLINE(performance-unnecessary-value-param)
190 ref_iterator(details::container_arg_t<T> container) :
191 container_{std::forward<details::container_arg_t<T>>(container)},
192 it_(std::begin(*container_.get())) {}
193
194 void swap(ref_iterator& other) noexcept {
195 std::swap(container_, other.container_);
196 std::swap(it_, other.it_);
197 std::swap(distance_, other.distance_);
198 rebase();
199 other.rebase();
200 }
201
202
203 ref_iterator& operator++() LIEF_LIFETIMEBOUND {
204 it_ = std::next(it_);
205 distance_++;
206 return *this;
207 }
208
209 ref_iterator operator++(int) {
210 ref_iterator retval = *this;
211 ++(*this);
212 return retval;
213 }
214
215 ref_iterator& operator--() LIEF_LIFETIMEBOUND {
216 if (it_ != std::begin(*container_.get())) {
217 it_ = std::prev(it_);
218 distance_--;
219 }
220 return *this;
221 }
222
223 ref_iterator operator--(int) {
224 ref_iterator retval = *this;
225 --(*this);
226 return retval;
227 }
228
229
230 ref_iterator& operator+=(
231 const typename ref_iterator::difference_type& movement
232 ) LIEF_LIFETIMEBOUND {
233 std::advance(it_, movement);
234 distance_ += movement;
235 return *this;
236 }
237
238
239 ref_iterator& operator-=(
240 const typename ref_iterator::difference_type& movement
241 ) LIEF_LIFETIMEBOUND {
242 return (*this) += -movement;
243 }

245
246 std::enable_if_t<!std::is_const_v<ref_t>, remove_const_t<ref_t>>
247 operator[](size_t n) {
248 return const_cast<remove_const_t<ref_t>>(
249 static_cast<const ref_iterator*>(this)->operator[](n)
250 );
251 }

253
254 add_const_t<ref_t> operator[](size_t n) const {
255 assert(n < size() && "integrity error: out of bound");
256 return details::deref<ref_t, DT_VAL>(std::next(std::begin(*container_.get()),
257 difference_type(n)));
258 }
259
260 ref_iterator operator+(typename ref_iterator::difference_type n) const {
261 ref_iterator tmp = *this;
262 return tmp += n;
263 }
264
265
266 ref_iterator operator-(typename ref_iterator::difference_type n) const {
267 ref_iterator tmp = *this;
268 return tmp -= n;
269 }
270
271
272 typename ref_iterator::difference_type operator-(const ref_iterator& rhs) const {
273 return distance_ - rhs.distance_;
274 }
275
276 bool operator<(const ref_iterator& rhs) const {
277 return (rhs - *this) > 0;
278 }
279
280
281 bool operator>(const ref_iterator& rhs) const {
282 return rhs < *this;
283 }
284
285
286 bool operator>=(const ref_iterator& rhs) const {
287 return !(*this < rhs);
288 }
289
290
291 bool operator<=(const ref_iterator& rhs) const {
292 return !(*this > rhs);
293 }
294
295 ref_iterator begin() const {
296 if (container_.get() == nullptr) {
297 return {};
298 }
299 return ref_iterator{*container_.get()};
300 }
301
302 ref_iterator cbegin() const {
303 return begin();
304 }
305
306 ref_iterator end() const {
307 ref_iterator it = begin();
308 if (auto* container = it.container_.get(); container != nullptr) {
309 it.it_ = std::end(*container);
310 it.distance_ = difference_type(container->size());
311 }
312 return it;
313 }
314
315 ref_iterator cend() const {
316 return end();
317 }
318
319 bool is_end() const {
320 return distance_ == difference_type(size());
321 }
322
323 bool operator==(const ref_iterator& other) const {
324 // If the current iterator borrows the container we assert the equality
325 // on the **address** of the container.
326 if constexpr (std::is_reference_v<T>) {
327 return container_.get() == other.container_.get() &&
328 distance_ == other.distance_;
329 } else {
330 return size() == other.size() && distance_ == other.distance_;
331 }
332 }
333
334 bool operator!=(const ref_iterator& other) const {
335 return !(*this == other);
336 }
337
338 size_t size() const {
339 const auto* container = container_.get();
340 return container != nullptr ? container->size() : 0;
341 }
342
343 bool empty() const {
344 const auto* container = container_.get();
345 return container == nullptr || container->empty();
346 }
347
348 std::enable_if_t<!std::is_const_v<ref_t>, remove_const_t<ref_t>> operator*() {
349 return const_cast<remove_const_t<ref_t>>(
350 static_cast<const ref_iterator*>(this)->operator*()
351 );
352 }
353
354 add_const_t<ref_t> operator*() const {
355 return details::deref<ref_t, DT_VAL>(it_);
356 }
357
358 std::enable_if_t<!std::is_const_v<pointer_t>, pointer_t> operator->() {
359 return const_cast<remove_const_t<pointer_t>>(
360 static_cast<const ref_iterator*>(this)->operator->()
361 );
362 }
363
364 add_const_t<pointer_t> operator->() const {
365 return const_cast<add_const_t<pointer_t>>(&(operator*()));
366 }
367
368 protected:
369 // If the container is owned by this iterator, make `it_` point in this
370 // container and not in the one it has been copied from
371 void rebase() {
372 if constexpr (!std::is_reference_v<T>) {
373 it_ = std::next(std::begin(*container_.get()), distance_);
374 }
375 }
376
377 details::container_storage<T> container_;
378 ITERATOR_T it_{};
379 typename ref_iterator::difference_type distance_{};
380};

391template<class T, typename U = typename decay_t<T>::value_type,
392 class ITERATOR_T = typename decay_t<T>::iterator>
393class filter_iterator {
394
395 public:
396 using iterator_category = std::forward_iterator_tag;
397 using iterator_concept =
398 details::iterator_concept_t<T, U, std::forward_iterator_tag>;
399 using value_type = std::remove_cv_t<std::remove_pointer_t<decay_t<U>>>;
400 using difference_type = ptrdiff_t;
401 using pointer = std::remove_pointer_t<U>*;
402 using reference = std::remove_pointer_t<U>&;
403
404 using container_type = T;
405 using DT_VAL = U;
406 using DT = decay_t<T>;
407 using ref_t = typename filter_iterator::reference;
408 using pointer_t = typename filter_iterator::pointer;
409 using filter_t = std::function<bool(const typename DT::value_type&)>;
410
413 filter_iterator() = default;
414
415 filter_iterator(const filter_iterator& other) :
416 size_c_{other.size_c_},
417 size_cached_{other.size_cached_},
418 container_{other.container_},
419 it_{other.it_},
420 filters_{other.filters_},
421 distance_{other.distance_} {
422 rebase();
423 }
424
425 filter_iterator(filter_iterator&& other) noexcept :
426 size_c_{other.size_c_},
427 size_cached_{other.size_cached_},
428 container_{std::move(other.container_)},
429 it_{other.it_},
430 filters_{std::move(other.filters_)},
431 distance_{other.distance_} {
432 rebase();
433 other.size_c_ = 0;
434 other.size_cached_ = false;
435 other.container_ = {};
436 other.it_ = {};
437 other.distance_ = 0;
438 }
439
440 filter_iterator& operator=(const filter_iterator& other) LIEF_LIFETIMEBOUND {
441 filter_iterator copy{other};
442 swap(copy);
443 return *this;
444 }
445
446 filter_iterator& operator=(filter_iterator&& other) noexcept LIEF_LIFETIMEBOUND {
447 swap(other);
448 return *this;
449 }
450
451 ~filter_iterator() = default;
452
453 // NOLINTNEXTLINE(performance-unnecessary-value-param)
454 filter_iterator(details::container_arg_t<T> container, filter_t filter) :
455 container_{std::forward<details::container_arg_t<T>>(container)},
456 it_(std::begin(*container_.get())),
457 filters_{} {
458 filters_.push_back(std::move(filter));
459
460 if (it_ != std::end(*container_.get()) && !match()) {
461 next();
462 }
463 }
464
465 // NOLINTNEXTLINE(performance-unnecessary-value-param)
466 filter_iterator(details::container_arg_t<T> container,
467 const std::vector<filter_t>& filters) :
468 container_{std::forward<details::container_arg_t<T>>(container)},
469 it_(std::begin(*container_.get())),
470 filters_{filters} {
471 if (it_ != std::end(*container_.get()) && !match()) {
472 next();
473 }
474 }
475
476 // NOLINTNEXTLINE(performance-unnecessary-value-param)
477 filter_iterator(details::container_arg_t<T> container) :
478 container_{std::forward<details::container_arg_t<T>>(container)},
479 it_(std::begin(*container_.get())),
480 filters_{} {}
481
482 void swap(filter_iterator& other) noexcept {
483 std::swap(container_, other.container_);
484 std::swap(it_, other.it_);
485 std::swap(filters_, other.filters_);
486 std::swap(size_c_, other.size_c_);
487 std::swap(size_cached_, other.size_cached_);
488 std::swap(distance_, other.distance_);
489 rebase();
490 other.rebase();
491 }
492
493
494 filter_iterator& def(filter_t func) LIEF_LIFETIMEBOUND {
495 filters_.push_back(func);
496 size_c_ = 0;
497 size_cached_ = false;
498 return *this;
499 }
500
501 filter_iterator& operator++() LIEF_LIFETIMEBOUND {
502 next();
503 return *this;
504 }
505
506 filter_iterator operator++(int) {
507 filter_iterator retval = *this;
508 ++(*this);
509 return retval;
510 }
511
512 filter_iterator begin() const {
513 if (container_.get() == nullptr) {
514 return {};
515 }
516 return {*container_.get(), filters_};
517 }
518
519 filter_iterator cbegin() const {
520 return begin();
521 }
522
523 filter_iterator end() const {
524 if (container_.get() == nullptr) {
525 return {};
526 }
527
528 // we don't need filter for the end iterator
529 filter_iterator it_end{*container_.get()};
530
531 it_end.it_ = std::end(*it_end.container_.get());
532 it_end.distance_ = difference_type(it_end.container_size());
533
534 return it_end;
535 }
536
537 filter_iterator cend() const {
538 return end();
539 }
540
541 bool is_end() const {
542 return distance_ == difference_type(container_size());
543 }
544
545 std::enable_if_t<!std::is_const_v<ref_t>, remove_const_t<ref_t>> operator*() {
546 return const_cast<remove_const_t<ref_t>>(
547 static_cast<const filter_iterator*>(this)->operator*()
548 );
549 }
550
551 add_const_t<ref_t> operator*() const {
552 return details::deref<ref_t, DT_VAL>(it_);
553 }
554
555
556 std::enable_if_t<!std::is_const_v<ref_t>, remove_const_t<ref_t>>
557 operator[](size_t n) {
558 return const_cast<remove_const_t<ref_t>>(
559 static_cast<const filter_iterator*>(this)->operator[](n)
560 );
561 }
562
563 add_const_t<ref_t> operator[](size_t n) const {
564 assert(n < size() && "integrity error: out of bound");
565
566 auto it = begin();
567 std::advance(it, n);
568
569 return details::deref<ref_t, DT_VAL>(std::next(std::begin(*container_.get()),
570 it.distance_));
571 }
572
573
574 std::enable_if_t<!std::is_const_v<pointer_t>, pointer_t> operator->() {
575 return const_cast<remove_const_t<pointer_t>>(
576 static_cast<const filter_iterator*>(this)->operator->()
577 );
578 }
579
580 add_const_t<pointer_t> operator->() const {
581 return const_cast<add_const_t<pointer_t>>(&(operator*()));
582 }
583
584 size_t size() const {
585 if (filters_.empty()) {
586 return container_size();
587 }
588
589 if (size_cached_) {
590 return size_c_;
591 }
592 filter_iterator it = begin();
593 size_t size = 0;
594
595 auto end_iter = it.end();
596 for (; it != end_iter; ++it) {
597 ++size;
598 }
599 size_c_ = size;
600 size_cached_ = true;
601 return size_c_;
602 }
603
604
605 bool empty() const {
606 if (filters_.empty()) {
607 return container_size() == 0;
608 }
609
610 if (size_cached_) {
611 return size_c_ == 0;
612 }
613
614 return begin() == end();
615 }
616
617
618 bool operator==(const filter_iterator& other) const {
619 if constexpr (std::is_reference_v<T>) {
620 return container_.get() == other.container_.get() &&
621 distance_ == other.distance_;
622 } else {
623 return container_size() == other.container_size() &&
624 distance_ == other.distance_;
625 }
626 }
627
628 bool operator!=(const filter_iterator& other) const {
629 return !(*this == other);
630 }
631
632 protected:
633 size_t container_size() const {
634 const auto* container = container_.get();
635 return container != nullptr ? container->size() : 0;
636 }
637
638 bool match() const {
639 return std::all_of(std::begin(filters_), std::end(filters_),
640 [this](const filter_t& f) { return f(*it_); });
641 }
642
643 void next() {
644 if (it_ == std::end(*container_.get())) {
645 distance_ = difference_type(container_size());
646 return;
647 }
648
649 do {
650 it_ = std::next(it_);
651 distance_++;
652 } while (it_ != std::end(*container_.get()) && !match());
653 }
654
655 void rebase() {
656 if constexpr (!std::is_reference_v<T>) {
657 it_ = std::next(std::begin(*container_.get()), distance_);
658 }
659 }
660
661
662 mutable size_t size_c_ = 0;
663 mutable bool size_cached_ = false;
664 details::container_storage<T> container_;
665 ITERATOR_T it_{};
666 std::vector<filter_t> filters_;
667 typename filter_iterator::difference_type distance_ = 0;
668};

701 class ReferenceProxy {

704 IteratorDecayTy it_;
705

708

711

713 return *it_;
714 }
715

744
745 private:
746 IteratorT begin_;
747 IteratorT end_;
748};
